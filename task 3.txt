CREATE DATABASE HR;

USE HR;

-- إنشاء جدول الموظفين
CREATE TABLE Employee (
  employee_id INT PRIMARY KEY,
  fname VARCHAR(100),
  lname VARCHAR(100),
  job_title VARCHAR(100),
  salary INT,
  hire_date DATE,
  department_id INT,
  CONSTRAINT fk_employee_department FOREIGN KEY (department_id) REFERENCES Department(department_id)
);

-- إنشاء جدول الأقسام
CREATE TABLE Department (
  department_id INT PRIMARY KEY,
  department_name VARCHAR(100),
  location_id INT,
  CONSTRAINT fk_department_location FOREIGN KEY (location_id) REFERENCES Location(location_id)
);

-- إنشاء جدول المواقع
CREATE TABLE Location (
  location_id INT PRIMARY KEY,
  city VARCHAR(100),
  country VARCHAR(100)
);

-- إنشاء جدول المشاريع
CREATE TABLE Project (
  project_id INT PRIMARY KEY,
  project_name VARCHAR(100),
  start_date DATE,
  end_date DATE
);

-- إنشاء جدول يربط الموظفين بالمشاريع (جدول وسيط لعلاقة متعدد لمتعدد)
CREATE TABLE Employee_Project (
  employee_id INT,
  project_id INT,
  PRIMARY KEY (employee_id, project_id),
  CONSTRAINT fk_employee_project_employee FOREIGN KEY (employee_id) REFERENCES Employee(employee_id),
  CONSTRAINT fk_employee_project_project FOREIGN KEY (project_id) REFERENCES Project(project_id)
);
